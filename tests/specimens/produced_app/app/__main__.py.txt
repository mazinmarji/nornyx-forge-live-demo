"""The support-case register's entry point: `python -m app`.

Written by the simulated provider of a Forge test. Forge's handoff check reads
this file and parses it; it never runs it.
"""

from __future__ import annotations

import argparse
import json

from app.cases import CaseRegister


def self_test() -> int:
    register = CaseRegister()
    ident = register.open("Printer offline", "high")
    register.assign(ident, "sam")
    register.resolve(ident)
    return 0 if register.case(ident)["state"] == "resolved" else 1


def main(argv: list[str] | None = None) -> int:
    parser = argparse.ArgumentParser(prog="app", description="A support-case register.")
    parser.add_argument("--self-test", action="store_true")
    arguments = parser.parse_args(argv)
    if arguments.self_test:
        return self_test()
    print(json.dumps({"application": "support-case register"}))
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
